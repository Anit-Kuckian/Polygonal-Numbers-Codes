# ============================================================
# EVEN POLYGONAL-NUMBER GENERATING SURFACES
#
# Change only these two inputs:
#
#   m     = an even integer with m >= 4
#   q_max = a prime or prime power
#
# The program calculates f(q) for every prime power q such that
#
#               2 <= q <= q_max.
# ============================================================

m = 64
q_max = 17


# ============================================================
# 1. INPUT VALIDATION
# ============================================================

def input_m(m):
    """
    Check that m is an even integer greater than or equal to 4.
    """

    m = ZZ(m)

    if m < 4 or m % 2 != 0:
        print(
            "The value of m must be greater than or equal to 4 "
            "and must be even."
        )
        return None

    return m


def input_q(q):
    """
    Check that q is a prime or prime power.
    """

    q = ZZ(q)

    if q < 2 or not q.is_prime_power():
        print(
            "The value of q must be greater than or equal to 2 "
            "and must be a prime or prime power."
        )
        return None

    return q


def input_q_max(q_max):
    """
    Check that the upper bound q_max is a prime or prime power.
    """

    q_max = ZZ(q_max)

    if q_max < 2 or not q_max.is_prime_power():
        print(
            "The value of q_max must be greater than or equal to 2 "
            "and must be a prime or prime power."
        )
        return None

    return q_max


# ============================================================
# 2. BLOCK CONTAINING m
# ============================================================

def _r_of(m):
    """
    Find the unique r >= 3 satisfying

        2^(r+1) + 4 <= m <= 2^(r+2) + 2.
    """

    m = input_m(m)

    if m is None:
        return None

    if m < 20:
        print("The value of m must be greater than or equal to 20.")
        return None

    r = 3

    while not (
        2^(r + 1) + 4 <= m <= 2^(r + 2) + 2
    ):
        r += 1

    return r


def A_of_m(m):
    """
    Return the unique block containing m.

        ("A1",)      represents A_1,
        ("A2",)      represents A_2,
        ("A", r, j)  represents A_{r,j}.
    """

    m = input_m(m)

    if m is None:
        return None

    if 4 <= m <= 10:
        return ("A1",)

    if 12 <= m <= 18:
        return ("A2",)

    r = _r_of(m)

    if r is None:
        return None

    j = (
        m - (2^(r + 1) + 4)
    ) // 8 + 1

    return ("A", r, j)


def block_entries(A):
    """
    Return all four entries of the block A.
    """

    if A is None:
        return None

    if A[0] == "A1":
        return (4, 6, 8, 10)

    if A[0] == "A2":
        return (12, 14, 16, 18)

    if A[0] != "A" or len(A) != 3:
        print(
            "The block must be A1, A2, "
            "or of the form ('A', r, j)."
        )
        return None

    r = A[1]
    j = A[2]

    first_entry = (
        8 * (2^(r - 2) + j - 1) + 4
    )

    return (
        first_entry,
        first_entry + 2,
        first_entry + 4,
        first_entry + 6
    )


def block_label(A):
    """
    Return a readable label for the block A.
    """

    if A is None:
        return None

    if A[0] == "A1":
        return "A_1"

    if A[0] == "A2":
        return "A_2"

    if A[0] != "A" or len(A) != 3:
        print(
            "The block must be A1, A2, "
            "or of the form ('A', r, j)."
        )
        return None

    return "A_{%d,%d}" % (
        A[1],
        A[2]
    )


def position_in_block(m):
    """
    Return the position of m in its block.
    """

    m = input_m(m)

    if m is None:
        return None

    A = A_of_m(m)

    if A is None:
        return None

    entries = block_entries(A)

    if entries is None:
        return None

    first_entry = entries[0]

    return (
        (m - first_entry) // 2 + 1
    )


# ============================================================
# 3. LENGTH OF THE POLYNOMIAL SYSTEM
# ============================================================

def n_e(m):
    """
    Return the length n_e(m) of the polynomial system.
    """

    m = input_m(m)

    if m is None:
        return None

    if 4 <= m <= 10:
        return 2

    if 12 <= m <= 18:
        return 3

    r = _r_of(m)

    if r is None:
        return None

    return r + 1


# ============================================================
# 4. PARENT BLOCK
# ============================================================

def Par(A):
    """
    Return the parent block of A.
    """

    if A is None:
        return None

    if A[0] == "A1":
        return None

    if A[0] == "A2":
        return ("A1",)

    if A[0] != "A" or len(A) != 3:
        print(
            "The block must be A1, A2, "
            "or of the form ('A', r, j)."
        )
        return None

    r = A[1]
    j = A[2]

    if j == 1:
        return ("A1",)

    if j == 2:
        return ("A2",)

    u = 3

    while not (
        2^(u - 2) < j <= 2^(u - 1)
    ):
        u += 1

    v = j - 2^(u - 2)

    return ("A", u, v)


# ============================================================
# 5. PARENT NUMBER
# ============================================================

def p_num(m):
    """
    Return the parent number p(m).

    The position of p(m) in the parent block is the same as the
    position of m in its own block.
    """

    m = input_m(m)

    if m is None:
        return None

    A = A_of_m(m)

    if A is None:
        return None

    if A[0] == "A1":
        return None

    parent_A = Par(A)

    if parent_A is None:
        return None

    position = position_in_block(m)

    if position is None:
        return None

    parent_entries = block_entries(parent_A)

    if parent_entries is None:
        return None

    return parent_entries[position - 1]


# ============================================================
# 6. ANCESTRY CHAINS
# ============================================================

def parent_chain(m):
    """
    Return the number ancestry in seed-to-target order:

        [m_0, m_1, ..., m_t],

    where m_0 belongs to A_1 and m_t = m.
    """

    m = input_m(m)

    if m is None:
        return None

    chain = [m]

    while p_num(chain[-1]) is not None:
        chain.append(
            p_num(chain[-1])
        )

    chain.reverse()

    return chain


def parent_block_chain(m):
    """
    Return the block ancestry in seed-to-target order.
    """

    number_chain = parent_chain(m)

    if number_chain is None:
        return None

    return [
        A_of_m(a)
        for a in number_chain
    ]


# ============================================================
# 7. EXTENSION NUMBER
# ============================================================

def nu(m):
    """
    Return

        nu(m) = n_e(m) - n_e(p(m)) - 1.
    """

    m = input_m(m)

    if m is None:
        return None

    parent = p_num(m)

    if parent is None:
        return None

    length_m = n_e(m)
    length_parent = n_e(parent)

    if length_m is None or length_parent is None:
        return None

    return (
        length_m
        - length_parent
        - 1
    )


# ============================================================
# 8. COMPACT POLYNOMIAL CODE
#
# Internal representations:
#
#   ("Q",)
#   ("R", depth)
#   ("P", depth)
#   ("sigma", seed, depth)
#
# depth = 0 means that no bracket indices occur.
# depth = 3 represents [1,2,3].
# ============================================================

def seed_code(a):
    """
    Return the seed code

        C(a) = (Q, sigma_a).
    """

    a = ZZ(a)

    if a not in [4, 6, 8, 10]:
        print("The seed must be 4, 6, 8, or 10.")
        return None

    return [
        ("Q",),
        ("sigma", a, 0)
    ]


def lift_code(C):
    """
    Apply one lift to the compact code.
    """

    if C is None:
        return None

    lifted = [
        ("Q",)
    ]

    for entry in C:

        tag = entry[0]

        if tag == "Q":
            lifted.append(
                ("R", 0)
            )

        elif tag == "R":
            lifted.append(
                ("R", entry[1] + 1)
            )

        elif tag == "P":
            lifted.append(
                ("P", entry[1] + 1)
            )

        elif tag == "sigma":
            lifted.append(
                (
                    "sigma",
                    entry[1],
                    entry[2] + 1
                )
            )

        else:
            print("Unknown compact-code entry.")
            return None

    return lifted


def one_extension_code(C):
    """
    Apply one 1-extension by appending P.
    """

    if C is None:
        return None

    return list(C) + [
        ("P", 0)
    ]


def iterated_extension_code(C, ell):
    """
    Apply ell successive 1-extensions.
    """

    if C is None:
        return None

    ell = ZZ(ell)

    if ell < 0:
        print("The extension number must be nonnegative.")
        return None

    D = list(C)

    for _ in range(ell):
        D = one_extension_code(D)

        if D is None:
            return None

    return D


def compact_code(m):
    """
    Construct the complete compact code C(m).
    """

    m = input_m(m)

    if m is None:
        return None

    chain = parent_chain(m)

    if chain is None:
        return None

    seed_value = chain[0]

    C = seed_code(seed_value)

    if C is None:
        return None

    for current_number in chain[1:]:

        C = lift_code(C)

        if C is None:
            return None

        extension_number = nu(current_number)

        if extension_number is None:
            return None

        C = iterated_extension_code(
            C,
            extension_number
        )

        if C is None:
            return None

    return C


# ============================================================
# 9. COMPACT CODE AS A STRING
# ============================================================

def index_string(depth):
    """
    Convert depth into bracket notation.

        depth = 0 gives ""
        depth = 1 gives "[1]"
        depth = 3 gives "[1,2,3]"
    """

    depth = ZZ(depth)

    if depth < 0:
        print("The depth must be nonnegative.")
        return None

    if depth == 0:
        return ""

    indices = ",".join(
        str(i)
        for i in range(1, depth + 1)
    )

    return "[" + indices + "]"


def code_entry_string(entry):
    """
    Convert one code entry into a readable string.
    """

    tag = entry[0]

    if tag == "Q":
        return "Q"

    if tag == "R":
        indices = index_string(entry[1])

        if indices is None:
            return None

        return "R" + indices

    if tag == "P":
        indices = index_string(entry[1])

        if indices is None:
            return None

        return "P" + indices

    if tag == "sigma":
        indices = index_string(entry[2])

        if indices is None:
            return None

        return (
            "sigma_%d" % entry[1]
            + indices
        )

    print("Unknown compact-code entry.")
    return None


def compact_code_string(m):
    """
    Return C(m) as a single readable string.
    """

    C = compact_code(m)

    if C is None:
        return None

    entries = []

    for entry in C:

        entry_string = code_entry_string(entry)

        if entry_string is None:
            return None

        entries.append(entry_string)

    return (
        "("
        + ", ".join(entries)
        + ")"
    )


# ============================================================
# 10. POLYNOMIAL NOTATION
# ============================================================

def notation_variables(depth, slot):
    """
    Return the variables occurring in a compact polynomial.

    For example:

        depth = 0, slot = 3

    gives

        Z,Y3,

    while

        depth = 2, slot = 4

    gives

        Z,Y1,Y2,Y4.
    """

    depth = ZZ(depth)
    slot = ZZ(slot)

    if depth < 0:
        print("The depth must be nonnegative.")
        return None

    if slot < 1:
        print("The Y-slot must be a positive integer.")
        return None

    variables = ["Z"]

    variables += [
        "Y%d" % i
        for i in range(1, depth + 1)
    ]

    variables.append(
        "Y%d" % slot
    )

    return ",".join(variables)


def polynomial_notation(entry, slot=None):
    """
    Convert a compact-code entry into polynomial notation.
    """

    tag = entry[0]

    if tag == "Q":
        return "Q(Z)"

    if slot is None:
        print("The Y-slot must be specified.")
        return None

    if tag == "R":

        depth = entry[1]
        order = depth + 1

        variables = notation_variables(
            depth,
            slot
        )

        if variables is None:
            return None

        return "R_%d(%s)" % (
            order,
            variables
        )

    if tag == "P":

        depth = entry[1]
        order = depth + 1

        variables = notation_variables(
            depth,
            slot
        )

        if variables is None:
            return None

        return "P_%d(%s)" % (
            order,
            variables
        )

    if tag == "sigma":

        seed = entry[1]
        depth = entry[2]

        variables = notation_variables(
            depth,
            slot
        )

        if variables is None:
            return None

        return "sigma_%d(%s)" % (
            seed,
            variables
        )

    print("Unknown compact-code entry.")
    return None


def polynomial_system_notation(m):
    """
    Return F^(m) in compact polynomial notation.
    """

    C = compact_code(m)

    if C is None:
        return None

    first_polynomial = polynomial_notation(C[0])

    if first_polynomial is None:
        return None

    notation = [
        first_polynomial
    ]

    for slot in range(1, len(C)):

        current_notation = polynomial_notation(
            C[slot],
            slot
        )

        if current_notation is None:
            return None

        notation.append(current_notation)

    return notation


# ============================================================
# 11. POINT-COUNT FUNCTION
# ============================================================

def polygonal_value(m, q):
    """
    Return the number of F_q-rational points:

        ((m-2)q^2 - (m-4)q)/2.
    """

    m = input_m(m)
    q = input_q(q)

    if m is None or q is None:
        return None

    return (
        (m - 2) * q^2
        - (m - 4) * q
    ) // 2


def f(q):
    """
    Return the number of F_q-rational points for the fixed m.

    The input q must be a prime or prime power.
    """

    global m

    return polygonal_value(m, q)


# ============================================================
# 12. PRIME POWERS IN THE REQUESTED RANGE
# ============================================================

def prime_powers_up_to(q_max):
    """
    Return every prime or prime power q satisfying

        2 <= q <= q_max.
    """

    q_max = input_q_max(q_max)

    if q_max is None:
        return None

    return [
        ZZ(q)
        for q in range(2, q_max + 1)
        if ZZ(q).is_prime_power()
    ]


def prime_power_description(q):
    """
    Return a readable expression for q = p^r.

    Examples:

        5  gives "5"
        8  gives "8 = 2^3"
        9  gives "9 = 3^2"
    """

    q = input_q(q)

    if q is None:
        return None

    p, r = q.is_prime_power(
        get_data=True
    )

    if r == 1:
        return str(q)

    return "%d = %d^%d" % (
        q,
        p,
        r
    )


# ============================================================
# 13. DISPLAY THE CONSTRUCTION
# ============================================================

def display_construction(m):
    """
    Display the structural information and polynomial system.
    """

    m = input_m(m)

    if m is None:
        return None

    A = A_of_m(m)

    if A is None:
        return None

    entries = block_entries(A)
    block_chain = parent_block_chain(m)
    number_chain = parent_chain(m)
    polynomial_list = polynomial_system_notation(m)

    if (
        entries is None
        or block_chain is None
        or number_chain is None
        or polynomial_list is None
    ):
        return None

    readable_block_label = block_label(A)

    if readable_block_label is None:
        return None

    print("=" * 78)
    print("EVEN POLYGONAL-NUMBER GENERATING SURFACE")
    print("=" * 78)
    print()

    print("Even number:")
    print("  m = %d" % m)
    print()

    print("Block containing m:")
    print(
        "  A(%d) = %s"
        % (
            m,
            readable_block_label
        )
    )
    print()

    print("Elements of the block:")
    print(" ", entries)
    print()

    print("Ancestry chain of blocks:")
    print(
        "  "
        + " -> ".join(
            block_label(B)
            for B in block_chain
        )
    )
    print()

    print("Ancestry chain of numbers:")
    print(
        "  "
        + " -> ".join(
            str(a)
            for a in number_chain
        )
    )
    print()

    print("Compact code:")
    print(
        "  C(%d) = %s"
        % (
            m,
            compact_code_string(m)
        )
    )
    print()

    print(
        "Polynomial system F^(%d):"
        % m
    )
    print()

    for i, notation in enumerate(
        polynomial_list,
        start=1
    ):
        print(
            "  F_%d = %s"
            % (
                i,
                notation
            )
        )

    print()

    return polynomial_list


# ============================================================
# 14. DISPLAY ALL VALUES f(q)
# ============================================================

def display_point_count_values(m, q_max):
    """
    Display f(q) for every prime power q from 2 through q_max.

    Return the results as a list of pairs

        [(q,f(q)), ...].
    """

    m = input_m(m)
    q_max = input_q_max(q_max)

    if m is None or q_max is None:
        return None

    q_values = prime_powers_up_to(q_max)

    if q_values is None:
        return None

    results = []

    for q in q_values:

        value = polygonal_value(m, q)

        if value is None:
            return None

        results.append(
            (
                q,
                value
            )
        )

    print("-" * 78)
    print(
        "VALUES OF f(q) FOR PRIME POWERS "
        "2 <= q <= %d" % q_max
    )
    print("-" * 78)
    print()

    print(
        "{:<18} {:>20}".format(
            "q",
            "f(q)"
        )
    )

    print(
        "{:<18} {:>20}".format(
            "-" * 18,
            "-" * 20
        )
    )

    for q, value in results:

        description = prime_power_description(q)

        if description is None:
            return None

        print(
            "{:<18} {:>20}".format(
                description,
                value
            )
        )

    print()
    print("=" * 78)

    return results


# ============================================================
# 15. AUTOMATIC EXECUTION
#
# The construction is displayed once. A for loop then evaluates
# f(q) at every prime or prime-power value up to q_max.
# ============================================================

m = input_m(m)
q_max = input_q_max(q_max)

if m is not None and q_max is not None:

    display_construction(m)

    point_count_values = display_point_count_values(
        m,

    )

else:
