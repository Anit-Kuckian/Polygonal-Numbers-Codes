from dataclasses import dataclass
from functools import lru_cache

# ============================================================
# SYMBOLIC CODE FOR CENTERED POLYGONAL-NUMBER FAMILIES
#
# Allowed entries:
#
#     Q0
#     Q
#     Q[1,...,s]
#     beta
#     beta[1,...,s]
#     Psi_a
#     Psi_a[1,...,s]
#
# where a in {4,6,8,10}.
# ============================================================

@dataclass(frozen=True)
class CodeSym:
    kind: str
    a: int = None
    J: tuple = ()


# ============================================================
# 1. INPUT
# ============================================================

def validate_m(m):
    m = int(m)

    if m < 3:
        raise ValueError("m must be an integer greater than or equal to 3.")

    return m


def associated_even_integer(m):
    """
    For odd m use the even family F^(m+1).
    """
    m = validate_m(m)

    if m % 2 == 0:
        return m

    return m + 1


# ============================================================
# 2. n(m), I_n, PARENT, nu(m)
# ============================================================

def n_value(m):
    """
    n(m) = min { n >= 2 : m <= 2^(n+1)+2 }.

    For even m >= 12:
        n(m) = floor(log_2(m-4)).
    """
    m = int(m)

    if m < 4 or m % 2 != 0:
        raise ValueError("n(m) is used here for even m >= 4.")

    if m <= 10:
        return 2

    return (m - 4).bit_length() - 1


def parent_even_number(m):
    """
    Par(m) = m - 2^(n(m)),  for m >= 12.
    """
    m = int(m)

    if m in (4, 6, 8, 10):
        return None

    n = n_value(m)

    return m - 2**n


def nu(m):
    """
    nu(m) = n(m) - n(Par(m)) - 1.
    """
    p = parent_even_number(m)

    if p is None:
        return None

    return n_value(m) - n_value(p) - 1


def parent_number_chain(m):
    """
    Returns
        m -> Par(m) -> ... -> m_0
    with m_0 in {4,6,8,10}.
    """
    m = int(m)

    chain = [m]

    while parent_even_number(chain[-1]) is not None:
        chain.append(parent_even_number(chain[-1]))

    return chain


# ============================================================
# 3. BLOCK INFORMATION A_{n,j}
# ============================================================

def block_position(m):
    """
    For m in I_n, n >= 3,

        m = 2^n + 4 + 8(j-1) + 2(k-1).

    Returns (n,j,k).

    For m in I_2, returns (2,1,k).
    """
    m = int(m)
    n = n_value(m)

    if n == 2:
        k = (m - 4)//2 + 1
        return (2, 1, k)

    r = (m - (2**n + 4)) // 2

    j = r // 4 + 1
    k = r % 4 + 1

    return (n, j, k)


def block_tuple(n, j):
    if n == 2:
        return (4, 6, 8, 10)

    a = 2**n + 4 + 8*(j - 1)

    return (a, a + 2, a + 4, a + 6)


def block_label(m):
    n, j, k = block_position(m)
    return f"A_{{{n},{j}}}"


def block_chain(m):
    return [block_label(a) for a in parent_number_chain(m)]


# ============================================================
# 4. INITIAL CODES
#
# C(4)  = (Q0, Psi_4)
# C(6)  = (Q0, Psi_6)
# C(8)  = (Q0, Psi_8)
# C(10) = (Q0, Psi_10)
# ============================================================

def initial_code(m):

    if m == 4:
        return (
            CodeSym("Q0"),
            CodeSym("Psi", 4)
        )

    if m == 6:
        return (
            CodeSym("Q0"),
            CodeSym("Psi", 6)
        )

    if m == 8:
        return (
            CodeSym("Q0"),
            CodeSym("Psi", 8)
        )

    if m == 10:
        return (
            CodeSym("Q0"),
            CodeSym("Psi", 10)
        )

    raise ValueError("Initial code only defined for 4,6,8,10.")


# ============================================================
# 5. THE L OPERATOR ON CODE ENTRIES
#
# Table 2:
#
# Q0                    -> Q
# Q                     -> Q[1]
# Q[1,...,s]            -> Q[1,...,s+1]
# beta                  -> beta[1]
# beta[1,...,s]         -> beta[1,...,s+1]
# Psi_a               -> Psi_a[1]
# Psi_a[1,...,s]      -> Psi_a[1,...,s+1]
# ============================================================

def lift_symbol(c):

    if c.kind == "Q0":
        return CodeSym("Q")

    if c.kind in ("Q", "beta", "Psi"):

        if len(c.J) == 0:
            new_J = (1,)
        else:
            new_J = tuple(range(1, len(c.J) + 2))

        return CodeSym(c.kind, c.a, new_J)

    raise ValueError("Unknown code symbol.")


def L_code(C):
    """
    C(L(F)) = (Q0, L(c1),...,L(cn)).
    """
    return (
        CodeSym("Q0"),
    ) + tuple(lift_symbol(c) for c in C)


# ============================================================
# 6. THE E OPERATOR
#
# E^ell(C) appends ell copies of beta.
# ============================================================

def E_code(C, ell=1):

    ell = int(ell)

    if ell < 0:
        raise ValueError("ell must be nonnegative.")

    return tuple(C) + tuple(
        CodeSym("beta")
        for _ in range(ell)
    )


# ============================================================
# 7. RECURSIVE CONSTRUCTION OF C(m)
#
# C(m) = E^{nu(m)}( L(C(Par(m))) )
# ============================================================

@lru_cache(None)
def even_code(m):

    m = int(m)

    if m < 4 or m % 2 != 0:
        raise ValueError("even_code requires an even m >= 4.")

    if m in (4, 6, 8, 10):
        return initial_code(m)

    p = parent_even_number(m)

    C_parent = even_code(p)

    C = L_code(C_parent)

    C = E_code(C, nu(m))

    return C


# ============================================================
# 8. SYMBOLIC STRING NOTATION
# ============================================================

def symbol_string(c):

    if c.kind == "Q0":
        return "Q0"

    if c.kind == "Q":
        base = "Q"

    elif c.kind == "beta":
        base = "beta"

    elif c.kind == "Psi":
        base = "Psi_" + str(c.a)

    else:
        raise ValueError("Unknown code symbol.")

    if len(c.J) == 0:
        return base

    indices = ",".join(str(j) for j in c.J)

    return base + "[" + indices + "]"


def even_code_string(m):

    C = even_code(m)

    return "(" + ", ".join(
        symbol_string(c) for c in C
    ) + ")"


def general_code_string(m):

    m = validate_m(m)

    e = associated_even_integer(m)

    C = even_code_string(e)

    if m % 2 == 0:
        return C

    return C + " ; Theta"


# ============================================================
# 9. POLYNOMIAL NOTATION
#
# Position t+1 determines final variable Y_t.
#
# Examples:
#
# Q at position 2:
#       Q_1(Y0,Y1)
#
# Psi_8[1] at position 3:
#       Psi_{8,2}(Y0,Y1,Y2)
#
# beta[1,2] at position 6:
#       beta_3(Y0,Y1,Y2,Y5)
# ============================================================

def indexed_arguments(c, final_index):

    args = ["Y_0"]

    args += [
        "Y_" + str(j)
        for j in c.J
    ]

    args.append("Y_" + str(final_index))

    return ",".join(args)


def polynomial_notation(c, position):

    # position is 1-based
    final_index = position - 1

    if c.kind == "Q0":
        return "Q_0(Y_0)"

    args = indexed_arguments(c, final_index)

    order = len(c.J) + 1

    if c.kind == "Q":
        return "Q_" + str(order) + "(" + args + ")"

    if c.kind == "beta":
        return "beta_" + str(order) + "(" + args + ")"

    if c.kind == "Psi":
        return (
            "Psi_" + str(c.a) + "," + str(order)
            + "(" + args + ")"
        )

    raise ValueError("Unknown code symbol.")


def polynomial_list(m):

    e = associated_even_integer(m)

    C = even_code(e)

    return [
        polynomial_notation(c, i)
        for i, c in enumerate(C, start=1)
    ]


# ============================================================
# 10. MACAULAY2 EXPANSION OF Q_s
#
# Q_0(U) = U^q-U
#
# Q_s(U0,...,Us)
#   = alpha(U0,Us)
#     + gamma(U0)Q_{s-1}(U1,...,Us)
# ============================================================

def expand_Q(arguments, V):

    U0 = arguments[0]

    if len(arguments) == 1:

        return (
            "("
            + "alpha(" + U0 + "," + V + ")"
            + "+"
            + "gamma(" + U0 + ")*Q0(" + V + ")"
            + ")"
        )

    tail = expand_Q(arguments[1:], V)

    return (
        "("
        + "alpha(" + U0 + "," + V + ")"
        + "+"
        + "gamma(" + U0 + ")*" + tail
        + ")"
    )


# ============================================================
# 11. MACAULAY2 EXPANSION OF beta_s
#
# beta_1(U0,U1)
#   = alpha(U0,U1)+gamma(U0)U1
#
# beta_s(U0,...,Us)
#   = alpha(U0,Us)
#     + gamma(U0)beta_{s-1}(U1,...,Us)
# ============================================================

def expand_beta(arguments, V):

    U0 = arguments[0]

    if len(arguments) == 1:

        return (
            "("
            + "alpha(" + U0 + "," + V + ")"
            + "+"
            + "gamma(" + U0 + ")*" + V
            + ")"
        )

    tail = expand_beta(arguments[1:], V)

    return (
        "("
        + "alpha(" + U0 + "," + V + ")"
        + "+"
        + "gamma(" + U0 + ")*" + tail
        + ")"
    )


# ============================================================
# 12. MACAULAY2 EXPANSION OF Psi_{a,s}
#
# Psi_{a,1}(U,V)=product_{j=1}^{(a-2)/2} rho_j(U,V).
# rho1=U*V-1; rho2=U-1; rho3=V;
# rho4=V-Delta(U-1).
# For s>=2 use alpha(U0,V)+gamma(U0)*Psi_{a,s-1}.
# ============================================================

def Psi_base(a, U, V):

    if a not in (4, 6, 8, 10):
        raise ValueError("a must belong to {4,6,8,10}.")

    factors = []
    for j in range(1, (a - 2)//2 + 1):
        factors.append("rho" + str(j) + "(" + U + "," + V + ")")

    return "(" + "*".join(factors) + ")"


def expand_Psi(a, arguments, V):

    U0 = arguments[0]

    if len(arguments) == 1:
        return Psi_base(a, U0, V)

    tail = expand_Psi(a, arguments[1:], V)

    return (
        "("
        + "alpha(" + U0 + "," + V + ")"
        + "+"
        + "gamma(" + U0 + ")*" + tail
        + ")"
    )


# ============================================================
# 13. CONVERT SYMBOLIC ENTRY TO MACAULAY2 POLYNOMIAL
# ============================================================

def polynomial_M2(c, position):

    final_index = position - 1

    if c.kind == "Q0":
        return "Q0(y0)"

    V = "y" + str(final_index)

    arguments = ["y0"] + [
        "y" + str(j)
        for j in c.J
    ]

    if c.kind == "Q":
        return expand_Q(arguments, V)

    if c.kind == "beta":
        return expand_beta(arguments, V)

    if c.kind == "Psi":
        return expand_Psi(c.a, arguments, V)

    raise ValueError("Unknown code symbol.")


def polynomial_list_M2(m):

    e = associated_even_integer(m)

    C = even_code(e)

    return [
        polynomial_M2(c, i)
        for i, c in enumerate(C, start=1)
    ]


# ============================================================
# 14. MACAULAY2 GENERATOR
#
# We choose the defining family
#
#       G_i = X Y_i - F_i.
#
# This is a valid choice from (3.3), with d_i=1 and R_i=0.
#
# For odd m we adjoin Theta(X,Y0).
# ============================================================

def macaulay2_code(m, good_qs=None):

    m = validate_m(m)

    odd_case = (m % 2 == 1)

    e = associated_even_integer(m)

    if good_qs is None:
        good_qs = [2, 3, 4, 5, 7]

    good_qs = [int(q) for q in good_qs]

    C = even_code(e)

    n = len(C)

    y_variables = [
        "y" + str(i)
        for i in range(n + 1)
    ]

    ring_variables = ",".join(
        ["z", "x"] + y_variables
    )

    polynomials = polynomial_list_M2(e)

    chain = parent_number_chain(e)

    lines = []

    lines.append("clearAll")
    lines.append("")
    lines.append("-- ====================================================")
    lines.append("-- CENTERED POLYGONAL-NUMBER FAMILY")
    lines.append("-- Section 7 symbolic-code construction")
    lines.append("-- ====================================================")
    lines.append("-- m = " + str(m))
    lines.append("-- associated even integer = " + str(e))
    lines.append("-- n(m_even) = " + str(n_value(e)))
    lines.append("-- C(" + str(e) + ") = " + even_code_string(e))
    lines.append("-- parent chain = " + " -> ".join(str(a) for a in chain))
    lines.append("-- block chain = " + " -> ".join(block_chain(e)))

    if odd_case:
        lines.append("-- odd case: adjoin Theta(X,Y0)")

    lines.append("")
    lines.append("f = qq -> (")
    lines.append("")
    lines.append("    K := GF(qq);")
    lines.append("")
    lines.append(
        "    S := K[" + ring_variables + ", MonomialOrder=>Lex];"
    )
    lines.append("")

    # Fundamental functions
    lines.append("    Q0 := U -> U^qq-U;")
    lines.append("    Delta := U -> U^(qq-1);")
    lines.append("    gamma := U -> 1-Delta(U);")
    lines.append(
        "    alpha := (U,V) -> Delta(U)*V*(V+U);"
    )
    lines.append("    rho1 := (U,V) -> U*V-1;")
    lines.append("    rho2 := (U,V) -> U-1;")
    lines.append("    rho3 := (U,V) -> V;")
    lines.append("    rho4 := (U,V) -> V-Delta(U-1);")
    lines.append("")

    # Associated family F_1,...,F_n
    for i, polynomial in enumerate(polynomials, start=1):
        lines.append(
            "    F" + str(i) + " := " + polynomial + ";"
        )

    # Theta for odd m
    if odd_case:

        lines.append("")
        lines.append("    pp := char K;")
        lines.append("    rr := 0;")
        lines.append("    fieldSize := 1;")
        lines.append("")

        lines.append("    while fieldSize < qq do (")
        lines.append("        fieldSize = fieldSize*pp;")
        lines.append("        rr = rr+1;")
        lines.append("    );")
        lines.append("")

        lines.append("    traceY0 := 0*y0;")
        lines.append("    ii := 0;")
        lines.append("")

        lines.append("    while ii < rr do (")
        lines.append(
            "        traceY0 = traceY0+y0^(pp^ii);"
        )
        lines.append("        ii = ii+1;")
        lines.append("    );")
        lines.append("")

        lines.append(
            "    LambdaQ := qq*(qq-1)//2;"
        )

        lines.append(
            "    eps2 := (2*x^0)^(qq-1);"
        )

        lines.append("")

        lines.append(
            "    Theta := "
            "eps2*x*(x^LambdaQ-1)"
            "+(1-eps2)*x^(qq-1)*traceY0;"
        )

    # --------------------------------------------------------
    # Base equations x*y_i-F_i, and Theta for odd m.
    # The z=0 branch is the base family; z=1 adds exactly
    # one rational point (all original coordinates zero).
    # --------------------------------------------------------

    base_equations = []
    for i in range(1, n + 1):
        base_equations.append("x*y" + str(i) + "-F" + str(i))

    if odd_case:
        base_equations.append("Theta")

    equations = ["z*(z-1)"]
    for R in base_equations:
        equations.append("(1-z)*(" + R + ")")

    equations.append("z*x")
    for i in range(n + 1):
        equations.append("z*y" + str(i))

    equations.append("z^qq-z")
    equations.append("x^qq-x")
    for i in range(n + 1):
        equations.append("y" + str(i) + "^qq-y" + str(i))

    lines.append("")
    lines.append("    I := ideal(")

    for i, equation in enumerate(equations):

        comma = "," if i < len(equations)-1 else ""

        lines.append(
            "        " + equation + comma
        )

    lines.append("    );")
    lines.append("")

    lines.append("    J := groebnerBasis I;")

    lines.append(
        "    computedCount := #flatten entries basis(S/ideal J);"
    )

    lines.append("")


    lines.append("")

    lines.append("    expectedCount := 1+(" + str(m) + "*qq*(qq-1))//2;")
    lines.append("    (qq,computedCount,expectedCount)")

    lines.append(")")
    lines.append("")

    # We need m available inside f
    # so replace symbolic mValue by literal m.
    text = "\n".join(lines)

    text = text.replace(
        "mValue",
        str(m)
    )

    q_string = "{" + ",".join(
        str(q) for q in good_qs
    ) + "}"

    text += "\n\n"
    text += "goodQs = " + q_string + ";\n"
    text += "results = apply(goodQs, qq -> f(qq));\n"
    text += "results"

    return text


# ============================================================
# 15. DISPLAY
# ============================================================

def show(m, good_qs=None):

    m = validate_m(m)

    odd_case = (m % 2 == 1)

    e = associated_even_integer(m)

    print("=" * 80)

    print("m =", m)

    print(
        "Case =",
        "odd" if odd_case else "even"
    )

    if odd_case:
        print(
            "Associated even integer =",
            e
        )

    print(
        "n(" + str(e) + ") =",
        n_value(e)
    )

    print(
        "Parent chain =",
        " -> ".join(
            str(a)
            for a in parent_number_chain(e)
        )
    )

    print(
        "Block chain =",
        " -> ".join(block_chain(e))
    )

    print("")

    print(
        "C(" + str(e) + ") =",
        even_code_string(e)
    )

    if odd_case:

        print(
            "Odd construction = C("
            + str(e)
            + ") together with Theta(X,Y0)"
        )

    print("=" * 80)

    print("")
    print("Associated centered family:")
    print("")

    for i, polynomial in enumerate(
        polynomial_list(e),
        start=1
    ):

        print(
            "F_" + str(i)
            + " = "
            + polynomial
        )

    if odd_case:

        print("")
        print(
            "Theta(X,Y_0) = "
            "2^(q-1) X(X^Lambda_q-1)"
            " + (1-2^(q-1)) X^(q-1) Tr(Y_0)"
        )

        print(
            "Lambda_q = q(q-1)/2"
        )

    print("")
    print("=" * 80)
    print("MACAULAY2 CODE")
    print("=" * 80)
    print("")

    try:
        print(
            macaulay2_code(
                m,
                good_qs=good_qs
            )
        )
    except ValueError as exc:
        print("Macaulay2 generation blocked:", exc)


# ============================================================
# 16. USER INPUT
# ============================================================

m = 15

good_qs = [2, 3, 4, 5, 7]

# Centered initial polynomials are built from rho1,rho2,rho3,rho4.


# ============================================================
# 17. RUN
# ============================================================

show(
    m,
    good_qs=good_qs
)
